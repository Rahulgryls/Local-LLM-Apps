# LAKO — Python Backend Dependencies
# Install: pip install -r requirements.txt
# Python version: 3.11+

# ── Web Framework ──────────────────────────────────────────
fastapi
uvicorn[standard]
python-multipart          # Required for file upload (UploadFile)

# ── Vector Database ────────────────────────────────────────
chromadb                  # Local vector storage — no Docker needed

# ── PDF Parsing ────────────────────────────────────────────
PyMuPDF                   # Primary PDF parser (fitz) — text + image extraction
pdfplumber                # Table extraction from native PDFs (bounding-box aware)
pytesseract               # OCR fallback for scanned PDFs

# ── Office Document Parsing ────────────────────────────────
python-docx               # Word document (.docx) extraction
openpyxl                  # Excel (.xlsx) reading
pandas                    # Excel sheet → DataFrame → text
python-pptx               # PowerPoint (.pptx) slide + notes extraction

# ── Ollama Client ──────────────────────────────────────────
ollama                    # Official Ollama Python client
httpx                     # Async HTTP client (used directly where needed)

# ── Confluence / Web ───────────────────────────────────────
aiofiles                  # Async file I/O for uploads
beautifulsoup4            # HTML → plain text (Confluence body parsing)

# ── Image Processing ───────────────────────────────────────
pillow                    # Image conversion for vision model inputs

# ── Hybrid Search ──────────────────────────────────────────
rank_bm25                 # BM25 keyword index — used alongside vector search (RRF fusion)

# ── Query Cache ────────────────────────────────────────────
cachetools                # TTLCache for LRU query result caching

# ── V2 Document Store ─────────────────────────────────────────
aiosqlite                 # Async SQLite access for V2 document store

# ── Structured Logging ────────────────────────────────────────
structlog                 # Structured logging for V2 pipeline

qdrant-client             # Qdrant vector DB client for V2 pipeline
